import { CommonModule } from '@angular/common';
import { Component, EventEmitter, Input, OnChanges, Output, SimpleChanges } from '@angular/core';
import { FormsModule } from '@angular/forms';

import {
  AccountPasswordForm,
  AccountProfileForm
} from '../../models/storefront.model';

@Component({
  selector: 'app-account-page',
  imports: [CommonModule, FormsModule],
  templateUrl: './account-page.component.html'
})
export class AccountPageComponent implements OnChanges {
  @Input() accountName: string | null = null;
  @Input({ required: true }) profileForm!: AccountProfileForm;
  @Input({ required: true }) passwordForm!: AccountPasswordForm;
  @Input() passwordHintMode = false;
  @Input() showPasswordHintMismatchNotice = false;
  @Input() profileErrorMessage = '';
  @Input() passwordErrorMessage = '';
  @Input() successMessage = '';

  @Output() backHome = new EventEmitter<void>();
  @Output() continueShopping = new EventEmitter<void>();
  @Output() requestSignIn = new EventEmitter<void>();
  @Output() saveProfile = new EventEmitter<void>();
  @Output() updatePassword = new EventEmitter<void>();
  @Output() openPasswordHintReset = new EventEmitter<void>();
  @Output() resetPasswordWithHint = new EventEmitter<void>();
  @Output() cancelPasswordHintReset = new EventEmitter<void>();

  profileEditMode = false;
  showProfileCurrentPassword = false;
  showAccountCurrentPassword = false;
  showAccountNewPassword = false;
  showAccountConfirmPassword = false;

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['accountName']?.currentValue === null) {
      this.profileEditMode = false;
      return;
    }

    if (changes['successMessage']?.currentValue === 'Account details updated successfully.') {
      this.profileEditMode = false;
    }
  }

  beginProfileEdit(): void {
    this.profileEditMode = true;
    this.profileForm.currentPassword = '';
  }

  profileNameErrorMessage(): string {
    return this.profileErrorMessage === 'Full name must be at least 3 letters.'
      ? this.profileErrorMessage
      : '';
  }

  profilePhoneErrorMessage(): string {
    return this.profileErrorMessage === 'Mobile number must be exactly 10 digits.'
      ? this.profileErrorMessage
      : '';
  }

  profileCurrentPasswordErrorMessage(): string {
    if (this.profileErrorMessage === 'Enter your current password before saving account details.') {
      return this.profileErrorMessage;
    }

    return this.profileErrorMessage ===
      'Current password is incorrect. Profile details were not updated.'
      ? this.profileErrorMessage
      : '';
  }

  passwordCurrentErrorMessage(): string {
    if (this.passwordHintMode) {
      return '';
    }

    if (this.passwordErrorMessage === 'Enter your current password first.') {
      return this.passwordErrorMessage;
    }

    return this.passwordErrorMessage.startsWith('Current password is incorrect')
      ? this.passwordErrorMessage
      : '';
  }

  passwordHintErrorMessage(): string {
    if (!this.passwordHintMode) {
      return '';
    }

    if (this.passwordErrorMessage === 'Enter the password hint saved for this account.') {
      return this.passwordErrorMessage;
    }

    return this.passwordErrorMessage === 'The password hint is incorrect for this account.'
      ? this.passwordErrorMessage
      : '';
  }

  newPasswordErrorMessage(): string {
    return this.passwordErrorMessage === 'New password must be at least 6 characters long.' ||
      this.passwordErrorMessage === 'New password must be different from the current password.'
      ? this.passwordErrorMessage
      : '';
  }

  confirmPasswordErrorMessage(): string {
    return this.passwordErrorMessage === 'Confirm password must match the new password.'
      ? this.passwordErrorMessage
      : '';
  }

  passwordActionErrorMessage(): string {
    return this.passwordErrorMessage === 'No password hint is saved for this account yet.'
      ? this.passwordErrorMessage
      : '';
  }

  togglePasswordVisibility(
    field: 'profileCurrent' | 'current' | 'new' | 'confirm'
  ): void {
    if (field === 'profileCurrent') {
      this.showProfileCurrentPassword = !this.showProfileCurrentPassword;
      return;
    }

    if (field === 'current') {
      this.showAccountCurrentPassword = !this.showAccountCurrentPassword;
      return;
    }

    if (field === 'new') {
      this.showAccountNewPassword = !this.showAccountNewPassword;
      return;
    }

    this.showAccountConfirmPassword = !this.showAccountConfirmPassword;
  }
}

      (this.selectedRecentOrderedProduct?.orderId ?? latestOrderId) !== latestOrderId

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['productForm']?.currentValue && this.productFormRef) {
      queueMicrotask(() => this.productFormRef?.resetForm(this.productForm));
    }

    if (
      changes['submitting']?.previousValue &&
      !changes['submitting'].currentValue &&
      this.successMessage &&
      this.isProductModalOpen
    ) {
      this.closeProductModal(false);
    }

    if (changes['overview']?.currentValue) {
      this.syncSupportReplyDrafts(this.supportTickets);

      if (
        this.activeTab === 'recent' &&
        !this.selectedOrderDetail &&
        !this.selectedProductDetail &&
        this.latestRecentOrderedProduct
      ) {
        queueMicrotask(() => this.openRecentOrder(this.latestRecentOrderedProduct!));
      }
    }

    if (
      changes['sectionSubmitting']?.previousValue &&
      !changes['sectionSubmitting'].currentValue &&
      this.successMessage &&
      this.isSectionModalOpen
    ) {
      this.closeSectionModal();
    }

    if (
      changes['subAdminSubmitting']?.previousValue &&
      !changes['subAdminSubmitting'].currentValue &&
      this.successMessage
    ) {
      this.closeSubAdminModal();
    }

    if (
      (changes['sections']?.currentValue || changes['categories']?.currentValue) &&
      this.catalogSectionFilter !== 'all' &&
      !this.categoryOptions.includes(this.catalogSectionFilter)
    ) {
      this.catalogSectionFilter = 'all';
    }

    if (
      changes['sections']?.currentValue &&
      this.editingSectionId !== null &&
      !this.managedSections.some((section) => section.id === this.editingSectionId)
    ) {
      this.editingSectionId = null;
      this.newSectionName = '';
    }

    if (changes['selectedOrderDetail']?.currentValue) {
      this.requestedOrderDetailId = this.selectedOrderDetail?.orderId ?? this.requestedOrderDetailId;
      this.requestedProductDetailId = null;
    }

    if (changes['selectedProductDetail']?.currentValue) {
      this.requestedProductDetailId = this.selectedProductDetail?.id ?? this.requestedProductDetailId;
      this.requestedOrderDetailId = null;
    }

    if (
      changes['isAuthenticated']?.currentValue &&
      typeof window !== 'undefined' &&
      this.normalizePath(window.location.pathname).startsWith('/admin')
    ) {
      this.applyAdminRouteFromLocation();
    }

    if (changes['adminUsers']) {
      this.clampCustomerUserPageIndex();

      if (
        this.selectedCustomerUserEmail &&
        !this.adminUsers.some((user) => user.email === this.selectedCustomerUserEmail)
      ) {
        this.closeCustomerUserModal(false);
      }
    }

    this.syncDocumentScrollLock();
  }

  requestSectionDelete(sectionId: number): void {
    if (!this.canDeleteCatalog || this.sectionSubmitting) {
      return;
    }

    const section = this.managedSections.find((item) => item.id === sectionId);

    if (!section) {
      return;
    }

    const productCount = this.getSectionAvailableProductCount(section.name);
    const confirmationMessage =
      productCount > 0
        ? `Delete the "${section.name}" section and ${productCount} ${
            productCount === 1 ? 'product' : 'products'
          } inside it?`
        : `Delete the "${section.name}" section?`;

    if (
      typeof window !== 'undefined' &&
      !window.confirm(confirmationMessage)
    ) {
      return;
    }

    this.deleteSection.emit(sectionId);
  }

    this.deleteProduct.emit(productId);

  requestCustomerUserDelete(user: AdminCustomerUser): void {
    if (!this.isMainAdmin || this.isDeletingCustomerUser(user.email)) {
      return;
    }

    const activitySummary = [
      user.orderCount ? `${user.orderCount} ${user.orderCount === 1 ? 'order' : 'orders'}` : '',
      user.ticketCount ? `${user.ticketCount} ${user.ticketCount === 1 ? 'support ticket' : 'support tickets'}` : ''
    ]
      .filter(Boolean)
      .join(' and ');
    const displayName = user.name || user.email;
    const confirmationMessage = activitySummary
      ? `Delete "${displayName}" from store users? This removes ${activitySummary} for ${user.email}.`
      : `Delete "${displayName}" from store users?`;

    if (typeof window !== 'undefined' && !window.confirm(confirmationMessage)) {
      return;
    }

    this.deleteCustomerUser.emit(user.email);
  }

  openCustomerUserModal(user: AdminCustomerUser): void {
    const normalizedEmail = String(user.email || '').trim().toLowerCase();

    if (!normalizedEmail) {
      return;
    }

    const requestId = ++this.customerUserOrdersRequestId;
    this.selectedCustomerUserEmail = normalizedEmail;
    this.selectedCustomerUserOrderId = null;
    this.customerUserOrders = [];
    this.customerUserOrdersLoading = true;
    this.customerUserOrdersError = '';
    this.syncDocumentScrollLock();

    this.storeService.getAdminCustomerUserOrders(normalizedEmail).subscribe({
      next: (orders) => {
        const normalizedOrders = Array.isArray(orders) ? [...orders] : [];

        if (!normalizedOrders.length && user.orderCount > 0) {
          this.loadCustomerUserOrdersFallback(normalizedEmail, requestId);
          return;
        }

        this.applyCustomerUserOrdersState(normalizedEmail, requestId, normalizedOrders);
      },
      error: (error: Error) => {
        if (user.orderCount > 0) {
          this.loadCustomerUserOrdersFallback(normalizedEmail, requestId, error.message);
          return;
        }

        this.applyCustomerUserOrdersState(normalizedEmail, requestId, [], error.message);
      }
    });
  }

  private scrollCatalogSectionToStart(): void {
    const sectionElement = this.catalogSectionStart?.nativeElement;

    if (!sectionElement) {
      return;
    }

    const toolbarHeight = this.catalogToolbar?.nativeElement.getBoundingClientRect().height ?? 0;
    const workspaceElement = this.adminWorkspace?.nativeElement;

    if (workspaceElement) {
      const workspaceTop = workspaceElement.getBoundingClientRect().top;
      const sectionTop = sectionElement.getBoundingClientRect().top;
      const targetTop = Math.max(
        0,
        workspaceElement.scrollTop + (sectionTop - workspaceTop) - toolbarHeight - 20
      );

      workspaceElement.scrollTo({ top: targetTop, behavior: 'smooth' });
      return;
    }

    if (typeof window === 'undefined') {
      return;
    }

    const sectionTop = window.scrollY + sectionElement.getBoundingClientRect().top;
    const targetTop = Math.max(0, sectionTop - toolbarHeight - 20);
    window.scrollTo({ top: targetTop, behavior: 'smooth' });
  }

  private scheduleCatalogSectionScroll(): void {
    if (typeof window !== 'undefined' && typeof window.requestAnimationFrame === 'function') {
      window.requestAnimationFrame(() => this.scrollCatalogSectionToStart());
      return;
    }

    queueMicrotask(() => this.scrollCatalogSectionToStart());
  }

  private applyAdminRouteFromLocation(): void {
    if (typeof window === 'undefined') {
      return;
    }

    const path = this.normalizePath(window.location.pathname);
    this.persistAdminRoute(path);

    if (!path.startsWith('/admin')) {
      return;
    }

    const segments = path.split('/').filter(Boolean);
    const section = segments[1] ?? 'dashboard';
    this.isApplyingAdminRoute = true;

    try {
      this.requestedOrderDetailId = null;
      this.requestedProductDetailId = null;
      this.detailOrigin = null;
      this.selectedRecentOrderedProduct = null;
      this.closeSupportTicketModal(false);
      this.closeProductModal(false, false);
      this.closeSectionModal(false, false);
      this.closeSubAdminModal(false, false);

      if (section === 'support' && segments[2] === 'ticket') {
        const ticketId = this.extractRouteNumber(segments[3]);

        if (ticketId) {
          this.setActiveTab('support', false);
          this.openSupportTicketModal(ticketId, false);
          return;
        }
      }

      if (section === 'products' && segments[2] === 'section') {
        if (segments[3] === 'new') {
          this.openNewSectionModal(false);
          return;
        }

        const sectionId = this.extractRouteNumber(segments[3]);

        if (sectionId && segments[4] === 'edit') {
          this.beginSectionEdit(sectionId, false);
          return;
        }
      }

      if (section === 'products' && segments[2] === 'product') {
        if (segments[3] === 'new') {
          this.openNewProductModal(false);
          return;
        }

        const productEditorId = this.extractRouteNumber(segments[3]);

        if (productEditorId && segments[4] === 'edit') {
          this.beginProductEdit(productEditorId, false);
          return;
        }
      }

      if (section === 'accounts' && segments[2] === 'sub-admin' && segments[3] === 'new') {
        this.setActiveTab('accounts', false);
        this.openSubAdminModal(false);
        return;
      }

      if (segments[2] === 'order') {
        const orderId = this.extractRouteNumber(segments[3]);
        const origin = this.mapPathSegmentToDetailOrigin(section);

        if (orderId && origin) {
          this.activeTab = this.getDisplayTabForOrderOrigin(origin);
          this.openOrderDetail(orderId, origin, false);
          return;
        }
      }

      if (segments[2] === 'product') {
        const productId = this.extractRouteNumber(segments[3]);
        const origin = this.mapPathSegmentToDetailOrigin(section);

        if (productId && origin) {
          this.activeTab = this.getDisplayTabForProductOrigin(origin);
          this.openProductDetail(productId, origin, false);
          return;
        }
      }

      this.setActiveTab(this.mapPathSegmentToAdminTab(section) ?? 'overview', false);
    } finally {
      this.isApplyingAdminRoute = false;
      this.syncDocumentScrollLock();
    }
  }

  private syncAdminRoute(replace = false): void {
    if (this.isApplyingAdminRoute || typeof window === 'undefined') {
      return;
    }

    const nextPath = this.buildAdminRoutePath();
    const currentPath = this.normalizePath(window.location.pathname);
    this.persistAdminRoute(nextPath);

    if (currentPath === nextPath) {
      return;
    }

    window.history[replace ? 'replaceState' : 'pushState']({}, '', nextPath);
  }

  private buildAdminRoutePath(): string {
    if (this.selectedSupportTicketId !== null) {
      return `/admin/support/ticket/${this.selectedSupportTicketId}`;

    if (this.isProductModalOpen) {
      return this.editingProductId !== null
        ? `/admin/products/product/${this.editingProductId}/edit`
        : '/admin/products/product/new';
    }

    if (this.isSectionModalOpen) {
      return this.editingSectionId !== null
        ? `/admin/products/section/${this.editingSectionId}/edit`
        : '/admin/products/section/new';
    }

    if (this.isSubAdminModalOpen) {
      return '/admin/accounts/sub-admin/new';
    }

    if (this.requestedOrderDetailId !== null) {
      return `/admin/${this.getDetailOriginPathSegment(this.detailOrigin)}/order/${this.requestedOrderDetailId}`;
    }

    if (this.requestedProductDetailId !== null) {
      return `/admin/${this.getDetailOriginPathSegment(this.detailOrigin)}/product/${this.requestedProductDetailId}`;
    }

    return `/admin/${this.getTabPathSegment(this.activeTab)}`;
  }

        );

    });
  }

    };
  }

import { CommonModule, CurrencyPipe } from '@angular/common';
import { Component, EventEmitter, Input, Output } from '@angular/core';
import { FormsModule, NgForm } from '@angular/forms';

import {
  CartItem,
  CheckoutForm,
  CheckoutStage,
  SavedDeliveryAddress
} from '../../models/storefront.model';

@Component({
  selector: 'app-checkout-page',
  imports: [CommonModule, FormsModule, CurrencyPipe],
  templateUrl: './checkout-page.component.html'
})
export class CheckoutPageComponent {
  @Input({ required: true }) stage!: CheckoutStage;
  @Input({ required: true }) checkoutForm!: CheckoutForm;
  @Input({ required: true }) paymentOptions!: string[];
  @Input({ required: true }) checkoutError!: string;
  @Input({ required: true }) orderSubmitting!: boolean;
  @Input({ required: true }) isCheckoutDisabled!: boolean;
  @Input({ required: true }) cart!: CartItem[];
  @Input({ required: true }) savedItems!: CartItem[];
  @Input() savedAddresses: SavedDeliveryAddress[] = [];
  @Input() selectedAddressId: string | null = null;
  @Input({ required: true }) checkoutItemCount!: number;
  @Input({ required: true }) cartSubtotal!: number;
  @Input({ required: true }) deliveryFee!: number;
  @Input({ required: true }) cartTotal!: number;

  @Output() submitOrder = new EventEmitter<void>();
  @Output() proceedCheckout = new EventEmitter<void>();
  @Output() backHome = new EventEmitter<void>();
  @Output() backToCart = new EventEmitter<void>();
  @Output() openProduct = new EventEmitter<CartItem>();
  @Output() increaseQuantity = new EventEmitter<number>();
  @Output() decreaseQuantity = new EventEmitter<number>();
  @Output() deleteItem = new EventEmitter<number>();
  @Output() saveForLater = new EventEmitter<number>();
  @Output() moveToCart = new EventEmitter<number>();
  @Output() deleteSavedItem = new EventEmitter<number>();
  @Output() chooseSavedAddress = new EventEmitter<string>();
  @Output() confirmSavedAddress = new EventEmitter<string>();
  @Output() editSavedAddress = new EventEmitter<string>();
  @Output() deleteSavedAddress = new EventEmitter<string>();
  @Output() startNewAddress = new EventEmitter<void>();

  get requiresCardDetails(): boolean {
    return this.checkoutForm.paymentMethod === 'Card payment';
  }

  handleSubmit(form: NgForm): void {
    if (form.invalid) {
      form.control.markAllAsTouched();
      return;
    }

    this.submitOrder.emit();
  }

  trackById(index: number, item: CartItem): number {
    return item.id;
  }

  getCartStockMessage(item: CartItem): string {
    const totalStock = Math.max(0, item.stock);
    const remainingStock = Math.max(0, totalStock - item.quantity);

    if (totalStock <= 0) {
      return 'Out of stock';
    }

    if (item.quantity >= totalStock) {
      return `All ${totalStock} available ${totalStock === 1 ? 'item is' : 'items are'} in your cart.`;
    }

    if (remainingStock <= 5) {
      return `Only ${remainingStock} ${remainingStock === 1 ? 'item is' : 'items are'} left in stock - order soon.`;
    }

    return 'In stock';
  }

  isCartStockWarning(item: CartItem): boolean {
    const remainingStock = Math.max(0, item.stock - item.quantity);
    return item.stock > 0 && item.quantity < item.stock && remainingStock <= 5;
  }

  selectPaymentMethod(option: string): void {
    this.checkoutForm.paymentMethod = option;

    if (option === 'Card payment') {
      return;
    }

    this.checkoutForm.cardHolderName = '';
    this.checkoutForm.cardNumber = '';
    this.checkoutForm.cardExpiry = '';
    this.checkoutForm.cardCvv = '';
  }
}

import { CommonModule, DOCUMENT } from '@angular/common';
import {
  Component,
  EventEmitter,
  HostListener,
  inject,
  Input,
  OnChanges,
  OnDestroy,
  Output,
  RendererFactory2,
  SimpleChanges
} from '@angular/core';

import {
  CustomerOrderDetail,
  CustomerOrderSummary
} from '../../models/order.model';

@Component({
  selector: 'app-orders-page',
  imports: [CommonModule],
  templateUrl: './orders-page.component.html',
  styleUrl: './orders-page.component.css'
})
export class OrdersPageComponent implements OnChanges, OnDestroy {
  private readonly document = inject(DOCUMENT);
  private readonly renderer = inject(RendererFactory2).createRenderer(null, null);

  @Input() accountName: string | null = null;
  @Input() orders: CustomerOrderSummary[] = [];
  @Input() ordersLoading = false;
  @Input() ordersError = '';
  @Input() selectedOrder: CustomerOrderDetail | null = null;
  @Input() selectedOrderLoading = false;
  @Input() selectedOrderError = '';
  @Input() activeOrderId: number | null = null;

  @Output() backHome = new EventEmitter<void>();
  @Output() continueShopping = new EventEmitter<void>();
  @Output() requestSignIn = new EventEmitter<void>();
  @Output() showOrderDetail = new EventEmitter<number>();
  @Output() closeOrderDetail = new EventEmitter<void>();

  ngOnChanges(_: SimpleChanges): void {
    this.syncPageScrollLock();
  }

  ngOnDestroy(): void {
    this.unlockPageScroll();
  }

  get isOrderDialogOpen(): boolean {
    return (
      this.activeOrderId !== null ||
      this.selectedOrder !== null ||
      this.selectedOrderLoading ||
      !!this.selectedOrderError
    );
  }

  openOrderDetail(orderId: number): void {
    this.showOrderDetail.emit(orderId);
  }

  closeSelectedOrderDetail(): void {
    this.closeOrderDetail.emit();
  }

  handleOrderCardKeydown(event: KeyboardEvent, orderId: number): void {
    if (event.key !== 'Enter' && event.key !== ' ') {
      return;
    }

    event.preventDefault();
    this.openOrderDetail(orderId);
  }

  @HostListener('document:keydown.escape')
  handleEscapeKey(): void {
    if (this.isOrderDialogOpen) {
      this.closeSelectedOrderDetail();
    }
  }

  getOrderStatusLabel(status: string): string {
    return status
      .split('_')
      .filter(Boolean)
      .map((part) => part.charAt(0).toUpperCase() + part.slice(1))
      .join(' ');
  }

  getOrderPreviewLabel(order: CustomerOrderSummary): string {
    if (order.itemCount <= 1) {
      return order.leadProductName;
    }

    return `${order.leadProductName} + ${order.itemCount - 1} more ${
      order.itemCount - 1 === 1 ? 'item' : 'items'
    }`;
  }

  private syncPageScrollLock(): void {
    if (this.isOrderDialogOpen) {
      this.lockPageScroll();
      return;
    }

    this.unlockPageScroll();
  }

  private lockPageScroll(): void {
    this.renderer.setStyle(this.document.documentElement, 'overflow', 'hidden');
    this.renderer.setStyle(this.document.body, 'overflow', 'hidden');
  }

  private unlockPageScroll(): void {
    this.renderer.removeStyle(this.document.documentElement, 'overflow');
    this.renderer.removeStyle(this.document.body, 'overflow');
  }
}

import { CommonModule } from '@angular/common';
import { Component, EventEmitter, Input, Output } from '@angular/core';
import { FormsModule, NgForm } from '@angular/forms';

import { AuthForm, AuthPasswordResetPayload } from '../../models/storefront.model';

@Component({
  selector: 'app-sign-in-page',
  imports: [CommonModule, FormsModule],
  templateUrl: './sign-in-page.component.html'
})
export class SignInPageComponent {
  @Input({ required: true }) form!: AuthForm;
  @Input({ required: true }) passwordResetForm!: AuthPasswordResetPayload;
  @Input() passwordResetMode: 'closed' | 'verify' | 'reset' = 'closed';
  @Input() noticeMessage = '';
  @Input() errorMessage = '';
  @Input() successMessage = '';

  @Output() submitForm = new EventEmitter<void>();
  @Output() openForgotPassword = new EventEmitter<void>();
  @Output() cancelForgotPassword = new EventEmitter<void>();
  @Output() submitPasswordResetVerification = new EventEmitter<void>();
  @Output() submitPasswordReset = new EventEmitter<void>();
  @Output() openSignUp = new EventEmitter<void>();
  @Output() openAdmin = new EventEmitter<void>();
  @Output() backHome = new EventEmitter<void>();

  showSignInPassword = false;
  showResetPassword = false;
  showResetConfirmPassword = false;

  handleSubmit(form: NgForm): void {
    if (this.passwordResetMode !== 'closed') {
      return;
    }

    if (form.invalid || !this.isIdentifierValid(this.form.identifier)) {
      form.control.markAllAsTouched();
      return;
    }

    this.submitForm.emit();
  }

  handleForgotPasswordClick(): void {
    this.openForgotPassword.emit();
  }

  handleCancelPasswordReset(): void {
    this.cancelForgotPassword.emit();
  }

  handlePasswordResetVerification(form: NgForm): void {
    if (form.invalid || !this.isIdentifierValid(this.passwordResetForm.identifier)) {
      form.control.markAllAsTouched();
      return;
    }

    this.submitPasswordResetVerification.emit();
  }

  handlePasswordReset(form: NgForm): void {
    if (!this.isPasswordResetValid()) {
      form.control.markAllAsTouched();
      return;
    }

    this.submitPasswordReset.emit();
  }

  isPasswordResetValid(): boolean {
    return (
      this.isIdentifierValid(this.passwordResetForm.identifier) &&
      this.passwordResetForm.newPassword.trim().length >= 6 &&
      this.passwordResetForm.newPassword.trim() ===
        this.passwordResetForm.confirmPassword.trim()
    );
  }

  isSignInStep(): boolean {
    return this.passwordResetMode === 'closed';
  }

  isVerifyStep(): boolean {
    return this.passwordResetMode === 'verify';
  }

  isResetStep(): boolean {
    return this.passwordResetMode === 'reset';
  }

  isIdentifierValid(value: string): boolean {
    const trimmedValue = value.trim();

    if (!trimmedValue) {
      return false;
    }

    if (trimmedValue.includes('@')) {
      return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(trimmedValue.toLowerCase());
    }

    return /^\d{10}$/.test(trimmedValue.replace(/\D/g, ''));
  }

  togglePasswordVisibility(field: 'signIn' | 'reset' | 'confirm'): void {
    if (field === 'signIn') {
      this.showSignInPassword = !this.showSignInPassword;
      return;
    }

    if (field === 'reset') {
      this.showResetPassword = !this.showResetPassword;
      return;
    }

    this.showResetConfirmPassword = !this.showResetConfirmPassword;
  }
}

import { CommonModule } from '@angular/common';
import {
  Component,
  EventEmitter,
  Input,
  OnChanges,
  Output,
  SimpleChanges,
  ViewChild
} from '@angular/core';
import { FormsModule, NgForm } from '@angular/forms';

import {
  CustomerSupportTicket,
  SupportForm,
  SupportTicketStatus
} from '../../models/support.model';
import { FaqItem, SupportCard } from '../../models/storefront.model';

@Component({
  selector: 'app-support-page',
  imports: [CommonModule, FormsModule],
  templateUrl: './support-page.component.html',
  styleUrl: './support-page.component.css'
})
export class SupportPageComponent implements OnChanges {
  @Input({ required: true }) supportCards!: SupportCard[];
  @Input({ required: true }) faqs!: FaqItem[];
  @Input({ required: true }) supportForm!: SupportForm;
  @Input({ required: true }) issueOptions!: string[];
  @Input() customerTickets: CustomerSupportTicket[] = [];
  @Input() ticketsLoading = false;
  @Input() ticketsError = '';
  @Input() accountName: string | null = null;
  @Input() errorMessage = '';
  @Input() successMessage = '';
  @Input() submitting = false;

  @Output() backHome = new EventEmitter<void>();
  @Output() continueBuilding = new EventEmitter<void>();
  @Output() submitForm = new EventEmitter<void>();
  @Output() requestSignIn = new EventEmitter<void>();
  @ViewChild('supportFormRef') supportFormRef?: NgForm;

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['successMessage']?.currentValue && this.supportFormRef) {
      queueMicrotask(() => this.supportFormRef?.resetForm(this.supportForm));
    }
  }

  handleSubmit(form: NgForm): void {
    if (!this.accountName) {
      this.requestSignIn.emit();
      return;
    }

    if (form.invalid) {
      form.control.markAllAsTouched();
      return;
    }

    this.submitForm.emit();
  }

  getSupportStatusLabel(status: SupportTicketStatus): string {
    const labels: Record<SupportTicketStatus, string> = {
      open: 'Open',
      in_progress: 'In progress',
      resolved: 'Resolved',
      closed: 'Closed'
    };

    return labels[status];
  }

  getSupportStatusMessage(ticket: CustomerSupportTicket): string {
    const messages: Record<SupportTicketStatus, string> = {
      open: 'We received your request and the support team will review it soon.',
      in_progress: 'The support team is actively working on this ticket.',
      resolved: 'The support team marked this issue as resolved. Review the reply above.',
      closed: 'This ticket is closed. Submit a new request if you still need help.'
    };

    return ticket.adminResponse || messages[ticket.status];
  }
}
